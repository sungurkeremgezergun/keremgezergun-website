# Content preferences for automated systems — https://contentsignals.org/
# Draft spec: draft-romm-aipref-contentsignals
#
#   ai-train  = training or fine-tuning AI models
#   search    = search applications that link back to this site
#   ai-input  = grounding / retrieval-augmented generation for AI answers
#
# Everything is allowed here on purpose. This site publishes SEO and GEO
# material and wants to be citable by AI answer engines, so ai-input=yes is
# deliberate — declaring it "no" would opt out of exactly the visibility the
# content is written for.
#
# The signal is repeated in every group rather than stated once: under the
# robots exclusion protocol a crawler reads only its single most specific
# matching group and inherits nothing from "*", so a named agent would never
# see a signal declared only there.

User-agent: *
Content-Signal: ai-train=yes, search=yes, ai-input=yes
Allow: /
